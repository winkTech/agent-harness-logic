```
examples/async_fifo-master/
├── rtl/                    # RTL源代码
│   ├── async_fifo.v        # 异步FIFO顶层模块
│   ├── async_bidir_fifo.v  # 双向FIFO
│   ├── async_bidir_ramif_fifo.v  # 带外部RAM的双向FIFO
│   ├── fifomem.v           # FIFO存储器
│   ├── fifomem_dp.v        # 双端口FIFO存储器
│   ├── rptr_empty.v        # 读指针和空标志
│   ├── wptr_full.v         # 写指针和满标志
│   ├── sync_ptr.v          # 指针同步
│   ├── sync_r2w.v          # 读到写时钟域同步
│   └── sync_w2r.v          # 写到读时钟域同步
├── sim/                    # 仿真测试
├── doc/                    # 文档
└── script/                 # 脚本
```
